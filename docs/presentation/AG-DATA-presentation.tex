\begin{frame}{1. What Are We Trying to Find?}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.485\textwidth}
      \begin{block}{Mandi: what price will be reported?}
        \small
        Forecast the modal price a mandi will report 1, 3, 7 and 14 days ahead: 36 commodities, 11,845 series.\\[0.3em]
        Each forecast has an 80\% conformal interval with measured coverage.\\[0.3em]
        Baseline: persistence (``last price''); 17--34\% of price changes are exactly zero.
      \end{block}
    \end{column}

    \begin{column}{0.485\textwidth}
      \begin{block}{Decision Support: outlook cards}
        \small
        Forecasts and intervals become cautious cards:
        \begin{itemize}\footnotesize
          \item \textbf{SELL:} even the upper interval estimate is not above the last price.
          \item \textbf{HOLD:} even the lower estimate exceeds the last price plus storage costs.
          \item \textbf{WATCH:} no clear signal (the default).
          \item \textbf{COMPARE:} a nearby mandi recently reported a clearly higher price.
        \end{itemize}
      \end{block}
    \end{column}
  \end{columns}

  \vspace{0.1cm}
  \begin{block}{Why do we need big data tools?}
    \small
    75.98 million archived reports and 21.06 million per-series window-feature rows do not fit single-node Pandas on a 16 GB laptop. HDFS stores the layers in replicated blocks; Spark splits the work across executor threads.
  \end{block}

  \vspace{0.05cm}
  \noindent{\small\textbf{Goal:} prepare the data, compare forecasting models honestly, and stream daily reports.}
\end{frame}

    \begin{column}{0.485\textwidth}
      \begin{block}{HDFS storage settings}
        \footnotesize
        \begin{itemize}
          \item 1 NameNode + 3 DataNodes (Docker, one host).
          \item Block size: 128 MiB; replication: 3 copies of each block.
          \item Silver: 692 MB, HEALTHY; readable with 1 DataNode down.
          \item Snappy Parquet, partitioned by \texttt{year=}.
        \end{itemize}
      \end{block}
    \end{column}

    \begin{column}{0.485\textwidth}
      \begin{block}{Spark application settings}
        \footnotesize
        \begin{itemize}
          \item Engine: Apache Spark 4.2 in \texttt{local[*]} mode (10 cores).
          \item Multi-core local CPU threads (NOT YARN worker executors).
          \item Shuffle partitions: 64 in the forecast jobs, coalesced by AQE.
          \item Also: Docker Spark cluster, 2 workers $\times$ 3 cores.
        \end{itemize}
      \end{block}
    \end{column}
  \end{columns}

  \vspace{0.15cm}
  \begin{block}{Are HDFS DataNodes also running as Spark executors?}
    \footnotesize
    No. HDFS (Docker) only stores blocks, replication 3; Spark runs in \texttt{local[*]} mode on host cores. The DataNodes share one disk: this shows HDFS behaviour, not hardware durability.
  \end{block}
\end{frame}

\begin{frame}{4. Spark Workflow and the DAG}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.485\textwidth}
      \begin{block}{Steps in Spark}
        \small
        \begin{enumerate}\footnotesize
          \item Read and parse CSVs with explicit StructType schemas.
          \item Quality rules and business-key de-duplication $\to$ silver.
          \item Window features per series ($\le t$) and targets ($> t$).
          \item Fit MLlib pipelines, choose on 2024, score the test once.
        \end{enumerate}
        \vspace{0.2em}
        {\footnotesize\textbf{\color{refRed}Partitions:} 64 shuffle partitions (AQE coalesces); \texttt{year=} directory pruning.}\\[0.2em]
        {\footnotesize\textbf{Scaling (measured):} full 75.6M rows, 10 cores: W1 42.2 s, W2 65.9 s; speed-up 2.4$\times$ / 3.0$\times$.}
      \end{block}
    \end{column}

    \begin{column}{0.485\textwidth}
      \begin{block}{DAG: a map of the work Spark runs}
        \footnotesize
        \textbf{Execution DAG (Shuffle-Delimited Stages):}
        \begin{center}
          \begin{tikzpicture}[node distance=0.18cm, auto, >=Stealth]
            \tikzstyle{sbox} = [rectangle, rounded corners=3pt, draw=refNavy, line width=0.8pt, fill=white, inner sep=3pt, minimum width=2.8cm, font=\tiny\bfseries, align=center]
            \node[sbox] (s0) {Parquet Read (silver)};

            \draw[->, line width=1pt, draw=refRed] (s0) -- (s1);
            \draw[->, line width=1pt, draw=refRed] (s1) -- (s2);
            \draw[->, line width=1pt, draw=refRed] (s2) -- (s3);
            \draw[->, line width=1pt, draw=refRed] (s3) -- (s4);

            \node[right=0.4cm of s2, align=left, font=\tiny, text width=3.2cm] {Wide transformations create shuffle boundaries.};
            \node[right=0.4cm of s4, align=left, font=\tiny, text width=3.2cm] {Exact stage counts are read from the Spark UI for each run.};
          \end{tikzpicture}
        \end{center}
      \end{block}
    \end{column}
  \end{columns}

  \vspace{0.15cm}
  \begin{block}{Where do the stages come from?}
    \footnotesize
    Narrow transformations pipeline within a stage. Wide transformations (\texttt{Window.partitionBy}, \texttt{groupBy}) need cross-partition shuffles, which create stage boundaries. Spark numbers stages at runtime.
  \end{block}
\end{frame}

\begin{frame}[fragile]{5. Scala Code and Streaming}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.50\textwidth}
      \begin{block}{Scala DataFrame API: source excerpt}
\begin{lstlisting}[language=Scala]
// 1. Non-fatal date parsing
expr("try_to_date(trim(Arrival_Date), 'yyyy-MM-dd')")
// 2. Business-key de-duplication
val w = Window.partitionBy(
    DataQuality.businessKey.map(col): _*)
  .orderBy(/* deterministic quality order */)
df.withColumn("_rn", row_number().over(w))
// 3. Leakage-safe features (Features.scala)
def days(lo: Long, hi: Long) = byEntity.rangeBetween(lo, hi)
.withColumn("mean30", avg(lp).over(days(-29, 0)))
\end{lstlisting}
      \end{block}
    \end{column}

    \begin{column}{0.47\textwidth}
      \begin{block}{What the code does}
        \footnotesize
        \begin{itemize}
          \item \textbf{Schema Enforcement:} explicit StructType schemas.
          \item \textbf{Resilient Parsing:} a malformed date becomes null, not a crash.
          \item \textbf{Deterministic Dedup:} one reproducible row per business key.
          \item \textbf{No Lookahead:} feature windows end at the current report; targets start after it; a perturbation test enforces this.
          \item \textbf{Shared Code:} batch and stream use one canonicalisation.
        \end{itemize}
      \end{block}
    \end{column}
  \end{columns}

  \vspace{0.0cm}
  \begin{block}{How do we use Structured Streaming?}
    \footnotesize
    Kafka topic \texttt{agri.prices.v1} $\to$ \texttt{PriceEventStream}: bronze (raw + offsets), quarantine before a 10-day watermark, de-dup by \texttt{event\_id}, silver and alerts. Effectively-once output per offset range.
  \end{block}
\end{frame}

\begin{frame}{6. How We Train and Test the Models}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.485\textwidth}
      \begin{block}{Baselines and models}
        \footnotesize
        \textbf{\color{refRed}Baselines}
        \begin{itemize}\scriptsize
          \item Persistence (last report), moving averages, seasonal naive.
          \item Target: log return to a report strictly after the origin.
        \end{itemize}
        \vspace{0.2em}
        \textbf{\color{refRed}Spark MLlib pipelines}
        \begin{itemize}\scriptsize
          \item Linear regression, random forest, gradient-boosted trees.
          \item One global model per horizon (next, 3, 7, 14 days).
        \end{itemize}
        \vspace{0.2em}
        {\scriptsize\color{refGreen}\textbf{Selected: GBT, absolute loss (next, 3 d); random forest, 60 trees, depth 10 (7, 14 d).}}
      \end{block}
    \end{column}

    \begin{column}{0.485\textwidth}
      \begin{block}{Strict temporal split \& tuning}
        \footnotesize
        \textbf{Chronological split (purged on target dates):}
        \begin{itemize}\scriptsize
          \item Train: origins 2016-01-01 .. 2023-12-31.
          \item Validation: 2024, model choice and conformal calibration.
          \item Test: 2025-01-01 .. 2026-04-20, scored once.
        \end{itemize}
        \vspace{0.2em}
        \textbf{Then, frozen models (no refitting):}
        \begin{itemize}\scriptsize
          \item Same test period on complete official data.
          \item Prospective 2026-04-22 $\to$ 2026-10-22: data that did not exist in the archive.
        \end{itemize}
      \end{block}
    \end{column}
  \end{columns}

  \vspace{0.15cm}
  \begin{block}{Why keep test data aside chronologically?}
    \footnotesize
    Random k-fold cross-validation lets future market information leak into training. Splitting chronologically and scoring the test once tests the models on genuinely later periods, as in deployment.
  \end{block}
\end{frame}

\begin{frame}{8. Challenges, Limits and Next Steps}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.485\textwidth}
      \begin{block}{Challenges we handled}
        \footnotesize
        \begin{itemize}
          \item JDK 21 reflection blocked Spark; solved via JVM \texttt{--add-opens} flags.
          \item Malformed dates handled by \texttt{try\_to\_date} instead of failing jobs.
          \item Persistence is hard to beat (17--34\% zero changes): absolute-loss GBT was needed.
          \item The archive mirror missed 70\% of recent reports; we back-filled official data.
        \end{itemize}
      \end{block}
    \end{column}

    \begin{column}{0.485\textwidth}
      \begin{block}{Limits and next steps}
        \footnotesize
        \begin{itemize}
          \item Local Spark (\texttt{local[*]}) and a one-host Docker cluster, not multi-node YARN.
          \item HDFS replicas share one disk; single NameNode (no HA).
          \item Gains over persistence are modest; per-series smoothing matches us at 1--3 days.
          \item No public deployment yet; decision-card hit rates are not farm profits.
        \end{itemize}
      \end{block}
    \end{column}
  \end{columns}

  \vspace{0.05cm}
  \begin{block}{What changed since v1, and why?}
    \footnotesize
    v1 reported an anomaly classifier with very high accuracy and Kafka speed figures. No code or log produces them, and the v1 labels leaked, so they are withdrawn (\texttt{docs/AUDIT\_PRIOR\_RESULTS.md}). We now lead with an honestly evaluated forecasting benchmark.
  \end{block}

  \vspace{0.05cm}
  \noindent{\small\textbf{Conclusion:} 3--8\% lower error than persistence at all horizons, calibrated 80\% intervals, and gains that hold on six unseen months.}
\end{frame}
